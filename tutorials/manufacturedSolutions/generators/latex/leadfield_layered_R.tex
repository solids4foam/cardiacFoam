$$
R = \begin{cases}
\frac{x - x_L}{\sigma_{b,2}} + \frac{x_L}{\sigma_{b,1}}, & x\le x_L,\\
\frac{x}{\sigma_{b,1}}, & x_L<x\le 0,\\
\frac{x}{(\boldsymbol\sigma_i+\boldsymbol\sigma_e)_{xx}}, & 0<x\le 1,\\
\frac{x - 1}{\sigma_{b,1}} + \frac{1}{(\boldsymbol\sigma_i+\boldsymbol\sigma_e)_{xx}}, & 1<x\le x_R,\\
\frac{x - x_R}{\sigma_{b,2}} + \frac{x_R - 1}{\sigma_{b,1}} + \frac{1}{(\boldsymbol\sigma_i+\boldsymbol\sigma_e)_{xx}}, & x>x_R.
\end{cases}
$$
