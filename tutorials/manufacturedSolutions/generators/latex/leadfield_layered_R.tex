$$
R = \begin{cases}
\frac{x - x_L}{\sigma_2} + \frac{x_L}{\sigma_1}, & x\le x_L,\\
\frac{x}{\sigma_1}, & x_L<x\le 0,\\
\frac{x}{(G_i+G_e)_{xx}}, & 0<x\le 1,\\
\frac{x - 1}{\sigma_1} + \frac{1}{(G_i+G_e)_{xx}}, & 1<x\le x_R,\\
\frac{x - x_R}{\sigma_2} + \frac{x_R - 1}{\sigma_1} + \frac{1}{(G_i+G_e)_{xx}}, & x>x_R.
\end{cases}
$$
