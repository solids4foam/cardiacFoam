% Keep a heading with the start of its content: a page break before a heading
% when less than the given space is left on the page, and no other break
% point added.
\usepackage{etoolbox}
\makeatletter
\newcommand{\mmskeepspace}[1]{\par
  \penalty\@M % brings \pagetotal up to date without adding a break point
  \begingroup
  \setlength{\dimen@}{#1}%
  \dimen@ii\pagegoal \advance\dimen@ii-\pagetotal
  % a full page (no space left) breaks earlier by itself
  \ifdim\dimen@ii>\z@
    \ifdim\dimen@>\dimen@ii \vfil\penalty-\@M \fi
  \fi
  \endgroup}
\makeatother
\pretocmd{\section}{\mmskeepspace{14\baselineskip}}{}{}
\pretocmd{\subsection}{\mmskeepspace{8\baselineskip}}{}{}
